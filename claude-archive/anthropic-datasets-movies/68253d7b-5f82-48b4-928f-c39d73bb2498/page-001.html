#!/usr/bin/env python3
"""Explode a CSV column containing a JSON (or Python-literal) array into one row per element.

Each element of the array becomes a row.  The element's keys become columns,
alongside whichever columns from the original row you asked to keep.

Example:
    ./explode_json_column.py credits.csv cast.csv --column cast --keep id
    ./explode_json_column.py credits.csv crew.csv --column crew --keep id --prefix crew_
"""

import argparse
import ast
import csv
import json
import sys

# Blobs in these files can be far larger than csv's default field limit.
csv.field_size_limit(sys.maxsize)


def parse_blob(raw):
    """Parse a cell holding an array.  Accepts strict JSON or a Python literal
    (single-quoted keys, None/True/False), which is how these dumps are written."""
    if raw is None:
        return []
    raw = raw.strip()
    if not raw or raw in ("[]", "nan", "NaN", "null", "None"):
        return []
    try:
        value = json.loads(raw)
    except (ValueError, TypeError):
        try:
            value = ast.literal_eval(raw)
        except (ValueError, SyntaxError, MemoryError, RecursionError):
            return None  # signals a parse failure to the caller
    if isinstance(value, dict):
        value = [value]
    if not isinstance(value, list):
        return None
    return value


def flatten(value):
    """Scalars pass through; nested containers are re-serialized as JSON."""
    if isinstance(value, (dict, list, tuple)):
        return json.dumps(value, ensure_ascii=False)
    if value is None:
        return ""
    return value


def iter_rows(path, encoding):
    with open(path, newline="", encoding=encoding, errors="replace") as fh:
        reader = csv.DictReader(fh)
        if reader.fieldnames is None:
            raise SystemExit(f"{path}: file is empty")
        yield reader.fieldnames
        yield from reader


def discover_keys(path, encoding, column, scan_rows):
    """First pass: union of element keys, in first-seen order."""
    rows = iter_rows(path, encoding)
    fieldnames = next(rows)
    if column not in fieldnames:
        raise SystemExit(
            f"{path}: column {column!r} not found; available: {', '.join(fieldnames)}"
        )
    keys = {}
    scalar_seen = False
    for n, row in enumerate(rows, 1):
        if scan_rows and n > scan_rows:
            break
        for element in parse_blob(row.get(column)) or []:
            if isinstance(element, dict):
                for key in element:
                    keys.setdefault(str(key), None)
            else:
                scalar_seen = True
    return fieldnames, list(keys), scalar_seen


def main():
    ap = argparse.ArgumentParser(description=__doc__,
                                 formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("input", help="input CSV")
    ap.add_argument("output", nargs="?", default="-", help="output CSV (default: stdout)")
    ap.add_argument("-c", "--column", required=True, help="column holding the JSON array")
    ap.add_argument("-k", "--keep", default="", metavar="COLS",
                    help="comma-separated columns from the original row to keep "
                         "(default: none; use 'all' for every other column)")
    ap.add_argument("--keys", default=None, metavar="KEYS",
                    help="comma-separated element keys to output; skips key discovery "
                         "and fixes column order")
    ap.add_argument("--scan-rows", type=int, default=0, metavar="N",
                    help="only scan the first N rows when discovering keys (default: all)")
    ap.add_argument("--prefix", default="", help="prefix for the exploded columns")
    ap.add_argument("--index-column", default=None, metavar="NAME",
                    help="also emit the element's position within its array")
    ap.add_argument("--encoding", default="utf-8")
    ap.add_argument("--keep-empty", action="store_true",
                    help="emit one blank row for rows whose array is empty")
    args = ap.parse_args()

    column = args.column
    scalar_seen = False

    if args.keys:
        keys = [k.strip() for k in args.keys.split(",") if k.strip()]
        fieldnames = next(iter_rows(args.input, args.encoding))
        if column not in fieldnames:
            raise SystemExit(
                f"{args.input}: column {column!r} not found; available: {', '.join(fieldnames)}"
            )
    else:
        fieldnames, keys, scalar_seen = discover_keys(
            args.input, args.encoding, column, args.scan_rows
        )

    if args.keep.strip().lower() == "all":
        keep = [c for c in fieldnames if c != column]
    else:
        keep = [c.strip() for c in args.keep.split(",") if c.strip()]
    missing = [c for c in keep if c not in fieldnames]
    if missing:
        raise SystemExit(
            f"{args.input}: --keep column(s) not found: {', '.join(missing)}; "
            f"available: {', '.join(fieldnames)}"
        )

    # Non-dict elements (plain strings/numbers in the array) go in a column of their own.
    value_col = f"{args.prefix}value" if scalar_seen else None
    out_keys = [f"{args.prefix}{k}" for k in keys]
    header = list(keep)
    if args.index_column:
        header.append(args.index_column)
    header += out_keys + ([value_col] if value_col else [])

    out = sys.stdout if args.output == "-" else open(
        args.output, "w", newline="", encoding="utf-8"
    )
    n_in = n_out = n_bad = n_empty = 0
    try:
        writer = csv.writer(out)
        writer.writerow(header)
        rows = iter_rows(args.input, args.encoding)
        next(rows)
        for row in rows:
            n_in += 1
            kept = [row.get(c, "") for c in keep]
            elements = parse_blob(row.get(column))
            if elements is None:
                n_bad += 1
                print(f"warning: row {n_in}: could not parse {column!r}, skipped",
                      file=sys.stderr)
                continue
            if not elements:
                n_empty += 1
                if args.keep_empty:
                    writer.writerow(kept + ([""] if args.index_column else "")
                                    + [""] * len(header[len(kept) + bool(args.index_column):]))
                    n_out += 1
                continue
            for i, element in enumerate(elements):
                out_row = list(kept)
                if args.index_column:
                    out_row.append(i)
                if isinstance(element, dict):
                    out_row += [flatten(element.get(k, "")) for k in keys]
                    if value_col:
                        out_row.append("")
                else:
                    out_row += [""] * len(keys)
                    if value_col:
                        out_row.append(flatten(element))
                writer.writerow(out_row)
                n_out += 1
    finally:
        if out is not sys.stdout:
            out.close()

    print(
        f"{n_in} input rows -> {n_out} output rows "
        f"({len(header)} columns, {n_empty} empty arrays, {n_bad} unparseable)",
        file=sys.stderr,
    )


if __name__ == "__main__":
    main()

Done. explode_json_column.py is a generic exploder, and I've run it on credits.csv.

Usage

./explode_json_column.py INPUT [OUTPUT] -c COLUMN -k KEEP_COLS [options]
  • -c/--column — the column holding the array
  • -k/--keep — comma-separated columns from the original row to carry along (all keeps every other column)
  • --prefix — prefix the exploded columns (needed here: cast elements have their own id/order that collide with the movie id)
  • --keys — skip discovery and fix the column set/order yourself
  • --scan-rows N — only scan N rows for key discovery (default: full first pass over the file)
  • --index-column NAME, --keep-empty, --encoding

It does a first pass to build the union of element keys in first-seen order, then a second streaming pass to write — so it never holds the file in memory. Notes on the implementation: the blobs are Python literals ('cast_id': 14, not JSON), so it tries json.loads then falls back to ast.literal_eval; nested dicts/lists inside an element are re-serialized as JSON into their cell; non-dict array elements land in a value column; csv.field_size_limit is raised since these fields are huge. It warns on duplicate output column names and on unparseable rows.
